\section{System Design: Cost Optimization}
% ============================================================

% ----------------------------------------------------------
\begin{frame}{Quota-Constrained Routing ($S_Q$ + $S_A$)}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Problem Setup:}
\begin{itemize}\setlength{\itemsep}{1pt}
    \item Providers: $S_Q$ (quota) + $S_A$ (API)
    \item $Q$ requests/day on $S_Q$ (zero cost)
    \item API cost: $c_i = \frac{n_i^{\text{in}}}{1000} p^{\text{in}} + \frac{n_i^{\text{out}}}{1000} p^{\text{out}}$
\end{itemize}

\vspace{0.1cm}
\begin{exampleblock}{Key Assumption: Fixed vs. Rolling}
\footnotesize
\begin{itemize}\setlength{\itemsep}{1pt}
    \item Real subscriptions often use \textbf{rolling windows}.
    \item We approximate this via \textbf{disjoint fixed intervals} (e.g., calendar day) where quota resets entirely.
    \item Yields an \textbf{optimistic lower bound} on API cost.
\end{itemize}
\end{exampleblock}

\vspace{0.1cm}
\textbf{Optimization:}
\vspace{-0.2cm}
\begin{equation*}
\min \sum_{i \in \mathcal{R}} x_i^A \cdot c_i^{\text{API}}
\end{equation*}
\vspace{-0.4cm}
\begin{equation*}
\text{s.t.} \quad \sum_{i: \text{day}(a_i)=d} x_i^Q \leq Q, \quad \forall d
\end{equation*}

\end{column}

\begin{column}{0.4\textwidth}
\begin{block}{Optimal Algorithm}
\footnotesize
\textbf{Closed-form solution:}
\begin{enumerate}\setlength{\itemsep}{1pt}
    \item For each day $d$, collect $\mathcal{R}_d$
    \item Sort by $c_i^{\text{API}}$ descending
    \item Top-$Q$ $\rightarrow$ $S_Q$
    \item Remainder $\rightarrow$ $S_A$
\end{enumerate}
\end{block}

\vspace{0.1cm}
\small
\textcolor{icmlgreen}{\textbf{Complexity:}} $O(n \log n)$

\vspace{0.1cm}
\textcolor{icmlblue}{\textbf{Intuition:}} Requests effectively ``bid'' for scarce quota using their avoided API cost.
\end{column}
\end{columns}
\end{frame}

% ----------------------------------------------------------
\begin{frame}{Joint Quota-Concurrency Routing ($S_Q$ + $S_C$ + $S_A$)}
\begin{columns}[T]
\begin{column}{0.58\textwidth}
\textbf{Additional Provider:}
\begin{itemize}
    \item $S_C$: Concurrency-limited subscription
    \item At most $C$ requests executing \textbf{simultaneously}
    \item Each request $i$ occupies a slot for $p_i$ seconds
\end{itemize}

\vspace{0.3cm}
\textbf{New Constraints:}
\begin{align*}
&\text{(Concurrency)} \quad \sum_{i} \mathbf{1}[s_i \leq t < s_i + p_i] \cdot x_i^C \leq C \\
&\text{(Arrival)} \quad s_i \geq a_i \\
&\text{(Deadline)} \quad s_i + p_i \leq d_i
\end{align*}

\end{column}

\begin{column}{0.38\textwidth}
\begin{alertblock}{Complexity Result}
Joint routing is \alert{\textbf{NP-hard}}\\[0.2cm]
\small
Couples:
\begin{itemize}
    \item Knapsack (daily quota)
    \item Parallel machine scheduling
    \item Release times + deadlines
\end{itemize}
\end{alertblock}

\vspace{0.3cm}
\begin{block}{Our Solution}
\textbf{Time-Indexed ILP}
\begin{itemize}
    \item Discretize into slots
    \item Use MIP solver (Gurobi)
    \item Exact optimal solution
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% ----------------------------------------------------------
\begin{frame}{Joint Routing: ILP Formulation}
\textbf{Time Discretization:}
\begin{itemize}
    \item Horizon: $t \in \{0, 1, \ldots, T\}$ with slot width $\Delta$
    \item Processing time: $\hat{p}_i = \lceil p_i / \Delta \rceil$ slots
    \item Valid window: $[\alpha_i, \beta_i]$ based on arrival and deadline
\end{itemize}

\vspace{0.3cm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Variables:}
\begin{itemize}
    \item $x_i^Q, x_i^C, x_i^A \in \{0,1\}$: routing
    \item $z_{i,t} \in \{0,1\}$: start at slot $t$
\end{itemize}

\vspace{0.2cm}
\textbf{Objective:}
\begin{equation*}
\min \sum_{i \in \mathcal{R}} x_i^A \cdot c_i^{\text{API}}
\end{equation*}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Key Constraints:}
\begin{align*}
&x_i^Q + x_i^C + x_i^A = 1 \\[0.2cm]
&\sum_{t=\alpha_i}^{\beta_i} z_{i,t} = x_i^C \\[0.2cm]
&\sum_i \sum_{\tau} z_{i,\tau} \leq C \quad \forall t
\end{align*}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Online Routing: Challenges \& Greedy Baseline}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Online Setting:} Requests arrive sequentially; routing decisions are \alert{irrevocable}.

\vspace{0.15cm}
\footnotesize
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Known:}
\begin{itemize}\setlength{\itemsep}{0pt}
    \item Model type $m_t$
    \item Input tokens $n_t^{\text{in}}$
    \item Remaining quota $q_t$
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Unknown:}
\begin{itemize}\setlength{\itemsep}{0pt}
    \item \alert{Output tokens} $n_t^{\text{out}}$
    \item \alert{Future requests}
\end{itemize}
\end{column}
\end{columns}

\vspace{0.2cm}
\small
\begin{algorithm}[H]
\caption{\small Greedy Strategy}
\begin{algorithmic}[1]
\footnotesize
\IF{$q_t > 0$}
    \RETURN SUBSCRIPTION
\ELSE
    \RETURN CHEAPEST\_API$(r_t)$
\ENDIF
\end{algorithmic}
\end{algorithm}
\end{column}

\begin{column}{0.48\textwidth}
\begin{exampleblock}{Cost-Aware Greedy Variant}
\footnotesize
\begin{algorithmic}[1]
\STATE $\hat{c}_t \gets$ EstimateCost$(r_t)$
\STATE $\theta \gets \frac{\text{Monthly Fee}}{30 \times Q}$
\IF{$\hat{c}_t > \theta$ \AND $q_t > 0$}
    \RETURN SUBSCRIPTION
\ELSE
    \RETURN API
\ENDIF
\end{algorithmic}
\end{exampleblock}

\vspace{0.15cm}
\begin{alertblock}{Greedy Limitations}
\small
\begin{itemize}\setlength{\itemsep}{1pt}
    \item Always uses quota first (FCFS)
    \item \alert{Wastes quota on cheap requests}
    \item Captures only \textbf{49--83\%} of optimal savings
\end{itemize}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Primal-Dual Threshold Strategy}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Core Insight:} Quota is a \alert{scarce resource}---assign it an \textbf{Opportunity Cost}!

\vspace{0.1cm}
\begin{block}{Shadow Price = Opportunity Cost}
\small
The threshold $\psi(z)$ represents the \textbf{marginal value} of the next quota slot:
\begin{equation*}
\psi(z) = L \cdot \left(\frac{U}{L}\right)^z
\end{equation*}
\begin{itemize}\setlength{\itemsep}{1pt}
    \item $z = k/Q$: fraction of quota consumed
    \item $L, U$: min/max request value bounds
    \item $\psi(z) \uparrow$ as quota becomes scarcer
\end{itemize}
\end{block}

\vspace{0.1cm}
\textbf{Decision Rule:}
\begin{equation*}
\text{Route to } S_Q \iff \underbrace{\hat{v}_t}_{\text{value}} \geq \underbrace{\psi(z)}_{\text{opp. cost}} \text{ and } k < Q
\end{equation*}
\end{column}

\begin{column}{0.42\textwidth}
\centering
\textbf{Adaptive Threshold}
\begin{tikzpicture}[scale=0.65]
\begin{axis}[
    xlabel={\small Quota Used ($z$)},
    ylabel={\small $\psi(z)$},
    xmin=0, xmax=1,
    ymin=0, ymax=1.2,
    grid=major,
    legend pos=north west,
    legend style={font=\tiny},
    width=6.5cm,
    height=5cm,
]
\addplot[domain=0:1, samples=50, thick, icmlblue] {0.05 * (20^x)};
\addlegendentry{$\psi(z) = L(U/L)^z$}
\addplot[domain=0:1, dashed, thick, icmlorange] {0.3};
\addlegendentry{Static threshold}
\node[circle, fill=icmlgreen, inner sep=2pt] at (axis cs:0,0.05) {};
\node[left, font=\tiny] at (axis cs:0,0.05) {$L$};
\node[circle, fill=icmlred, inner sep=2pt] at (axis cs:1,1.0) {};
\node[right, font=\tiny] at (axis cs:1,1.0) {$U$};
\end{axis}
\end{tikzpicture}

\vspace{0.1cm}
\begin{alertblock}{Theoretical Guarantee}
\small
Competitive ratio: $O(\ln(U/L) + 1)$\\
(Optimal for online knapsack with prices)
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Primal-Dual: Handling Unknown Output Length}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{The Reality:} Output length is unknown at routing time!
\begin{itemize}\setlength{\itemsep}{2pt}
    \item We cannot compute true value $v_t$
    \item We must predict $\hat{v}_t$ using historical distributions
\end{itemize}

\vspace{0.2cm}
\begin{block}{Output Length Predictors}
\small
\begin{itemize}\setlength{\itemsep}{2pt}
    \item \textbf{Histogram:} Maintains full empirical CDF.\\
    \textcolor{icmlgreen}{Pros:} Well-calibrated (accurate P50/P10).\\
    \textcolor{icmlred}{Cons:} Slow to adapt to workload shifts.
    \item \textbf{EMA (Exponential Moving Average):}\\
    $\hat{n}_{\text{new}} = \alpha \cdot n_t + (1-\alpha) \cdot \hat{n}_{\text{old}}$\\
    \textcolor{icmlgreen}{Pros:} Adapts in $\sim$10 samples.\\
    \textcolor{icmlred}{Cons:} Point estimate, statistical bias.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{exampleblock}{Routing with Predictions}
\small
\begin{algorithmic}[1]
\STATE $\hat{n}_t^{\text{out}} \gets$ Predictor.predict()
\STATE $\hat{v}_t \gets \text{Price}_{\text{in}} \cdot n_t^{\text{in}} + \text{Price}_{\text{out}} \cdot \hat{n}_t^{\text{out}}$
\STATE $\theta \gets \psi(k/Q)$ \COMMENT{Shadow Price}
\IF{$\hat{v}_t \geq \theta$ \AND $k < Q$}
    \RETURN SUBSCRIPTION
\ELSE
    \RETURN API
\ENDIF
\end{algorithmic}
\end{exampleblock}

\vspace{0.2cm}
\fbox{\parbox{0.9\textwidth}{
\centering
\small
\textbf{Risk:} What if EMA over-predicts length?\\
We waste scarce quota on a cheap request!
}}
\end{column}
\end{columns}
\end{frame}

        \vspace{0.3cm}
        \small
        \textbf{LCB (Lower Confidence Bound):} Use P10 to avoid\\
        over-estimating value $\to$ prevents quota waste.
    \end{column}

    \begin{column}{0.48\textwidth}
        \begin{exampleblock}{Risk-Aware Decision Rule}
        \begin{algorithmic}[1]
        \STATE $\hat{v}_{\text{LCB}} \gets$ Predictor.P10$(r_t)$
        \STATE $\theta \gets \psi(k/Q)$ \COMMENT{Shadow Price}

        \IF{$\hat{v}_{\text{LCB}} \geq \theta$ \AND $k < Q$}
            \RETURN SUBSCRIPTION
        \ELSE
            \RETURN API
        \ENDIF
        \end{algorithmic}
        \end{exampleblock}

        \vspace{0.2cm}
        \begin{alertblock}{Why It Works}
        \begin{itemize}
            \item \textbf{High Variance?} LCB $\ll$ Median\\
            $\to$ Uncertain value $\to$ Use API (safe)
            \item \textbf{Low Variance?} LCB $\approx$ Median\\
            $\to$ Confident value $\to$ Use $S_Q$
        \end{itemize}
        \end{alertblock}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Worked Example: Primal-Dual vs Greedy}
\small
\textbf{Setup:} Quota $Q=3$, value bounds $[L,U]=[\$0.10, \$1.00]$. Five requests arrive:

\vspace{0.1cm}
\footnotesize
\begin{columns}[T]
\begin{column}{0.35\textwidth}
\begin{tabular}{@{}ccc@{}}
\toprule
Req & Out Tok & Cost \\
\midrule
$r_1$ & 200 & \$0.15 \\
$r_2$ & 800 & \$0.45 \\
$r_3$ & 100 & \$0.08 \\
$r_4$ & 1500 & \$0.85 \\
$r_5$ & 600 & \$0.38 \\
\bottomrule
\end{tabular}
\end{column}

\begin{column}{0.62\textwidth}
\textbf{Greedy:} Accept in order $\to r_1, r_2, r_3$ to $S_Q$\\
API cost: $\$0.85 + \$0.38 = \$1.23$. Savings: \$0.68

\vspace{0.15cm}
\textbf{Primal-Dual:} $\psi(z) = 0.10 \cdot 10^z$
\begin{itemize}\setlength{\itemsep}{0pt}
    \item $r_1$: $z\!=\!0$, $\psi\!=\!\$0.10$. \$0.15 $>$ \$0.10 \textcolor{icmlgreen}{Accept}
    \item $r_2$: $z\!=\!0.33$, $\psi\!=\!\$0.22$. \$0.45 $>$ \$0.22 \textcolor{icmlgreen}{Accept}
    \item $r_3$: $z\!=\!0.67$, $\psi\!=\!\$0.46$. \$0.08 $<$ \$0.46 \textcolor{icmlred}{Reject}
    \item $r_4$: $z\!=\!0.67$, $\psi\!=\!\$0.46$. \$0.85 $>$ \$0.46 \textcolor{icmlgreen}{Accept}
    \item $r_5$: Quota full $\to$ API
\end{itemize}
API cost: $\$0.08 + \$0.38 = \$0.46$. Savings: \$1.45
\end{column}
\end{columns}

\vspace{0.15cm}
\begin{center}
\fbox{\parbox{0.8\textwidth}{
\centering
\textbf{Primal-Dual saves 2.1x more} by rejecting cheap $r_3$ (\$0.08) and reserving quota for expensive $r_4$ (\$0.85).
}}
\end{center}
\end{frame}

\begin{frame}{Online Adaptive Routing}
\vspace{-0.3cm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\footnotesize
\textbf{Challenge:} Latency distributions are \alert{non-stationary}
\begin{itemize}\setlength{\itemsep}{0pt}
    \item Provider load varies by time-of-day
    \item Need \textbf{adaptive} profiling
\end{itemize}

\vspace{0.15cm}
\textbf{Moving Window Profiler:}
\begin{itemize}\setlength{\itemsep}{0pt}
    \item Short (15 min) + Long (3h) windows
    \item Adaptive: $\beta = \frac{N_{\text{short}}}{N_{\text{short}} + \lambda}$
\end{itemize}
\[
\hat{F}(L) = \beta \cdot F_{\text{short}} + (1-\beta) \cdot F_{\text{long}}
\]
\end{column}

\begin{column}{0.48\textwidth}
\footnotesize
\textbf{Online Algorithm:}
\begin{enumerate}\setlength{\itemsep}{0pt}
    \item Update profiles with recent samples
    \item Solve LP: $\pi^* = \arg\min_\pi \sum_j \pi_j c_j$
    \item Sample provider via SWRR($\pi^*$)
\end{enumerate}

\vspace{0.15cm}
\textbf{SWRR = Smooth Weighted Round-Robin}
\begin{itemize}\setlength{\itemsep}{0pt}
    \item Deterministic sampling from $\pi^*$
    \item Exact traffic proportions
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Smart Hedging: Cost-Benefit Rule}
\vspace{-0.3cm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\scriptsize
\textbf{Problem:} Tail latency causes SLO violations even with optimal routing.

\vspace{0.1cm}
\textbf{Cost-benefit rule:} Hedge when expected benefit $>$ backup cost:
$$\underbrace{\frac{S_{primary}(L)}{S_{primary}(t)}}_{\text{Primary Risk}} \times \underbrace{F_{backup}(Rem)}_{\text{Backup Viability}} > \underbrace{\frac{C_b}{V}}_{\text{Cost Ratio}}$$

\vspace{0.1cm}
\centering
\begin{tikzpicture}[scale=0.55]
    \draw[->, thick] (0,0) -- (5,0) node[right] {\tiny Time};
    \draw[thick, icmlblue] (0,1) -- (3.8,1);
    \node[left, font=\tiny] at (0,1) {Primary};
    \filldraw[icmlblue] (3.8,1) circle (2pt);
    \draw[thick, icmlred, dashed] (1.5,0) -- (1.5,1.4);
    \node[above, font=\tiny, text=icmlred] at (1.5,1.4) {Hedge};
    \draw[thick, icmlorange] (1.5,0.3) -- (2.8,0.3);
    \node[left, font=\tiny] at (0,0.3) {Backup};
    \filldraw[icmlorange] (2.8,0.3) circle (2pt);
    \draw[thick, icmlgreen, dashed] (4.2,0) -- (4.2,1.4);
    \node[above, font=\tiny, text=icmlgreen] at (4.2,1.4) {SLO};
\end{tikzpicture}
\end{column}

\begin{column}{0.48\textwidth}
\scriptsize
\begin{exampleblock}{Key Properties}
\begin{itemize}\setlength{\itemsep}{1pt}
    \item \textbf{Mathematically Rigorous:} Evaluates joint probability that \textit{both} models violate SLO.
    \item \textbf{Economically Interpretable:} Hedge only if expected benefit $>$ backup cost.
    \item \textbf{Natural Adaptivity:} Automatically hedges more aggressively if backup is cheap ($C_b \downarrow$), and more conservatively if backup is expensive ($C_b \uparrow$).
\end{itemize}
\end{exampleblock}

\vspace{0.1cm}
\begin{tabular}{lccc}
\toprule
Strategy & Viol. & Hedge & $C_b/V$ \\
\midrule
No hedging & 1.4\% & 0\% & --- \\
Fixed 0.5 & 0\% & 5.2\% & --- \\
Survival & 0\% & 15.2\% & --- \\
\textbf{Economic} & \textbf{0\%} & \textbf{8.6\%} & \textbf{0.05} \\
\bottomrule
\end{tabular}

\end{column}
\end{columns}
\end{frame}

\begin{frame}[noframenumbering]
\hypertarget{app:ilp}{}{Appendix B: Time-Indexed ILP Details}
\textbf{Concurrency Constraint (Linearized):}
\begin{equation*}
\sum_{i \in \mathcal{R}} \sum_{\tau=\max(\alpha_i, t-\hat{p}_i+1)}^{\min(\beta_i, t)} z_{i,\tau} \leq C, \quad \forall t \in \{0, \ldots, T\}
\end{equation*}

\vspace{0.3cm}
\textbf{Variable Count:}
\begin{itemize}
    \item Routing: $3n$ binary variables
    \item Scheduling: $O(n \cdot T / \Delta)$ binary variables
    \item Trade-off: finer $\Delta$ $\Rightarrow$ more accurate but larger problem
\end{itemize}
\end{frame}

% D. Why PrimalDual Beats ILP
% ----------------------------------------------------------
\begin{frame}[noframenumbering]
\hypertarget{app:q1}{}{Appendix C: Why PrimalDual Beats the ILP Oracle}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{ILP Time Discretization:}
\begin{itemize}\setlength{\itemsep}{2pt}
    \item Time divided into $\Delta=1$s slots
    \item Request finishing at 0.3s $\to$ slot occupied until 1.0s
    \item \alert{Capacity fragmentation}: 0.7s wasted per slot
\end{itemize}

\vspace{0.2cm}
\textbf{PrimalDual (Continuous-Time):}
\begin{itemize}\setlength{\itemsep}{2pt}
    \item Event-driven: acts at exact arrival/completion times
    \item No wasted capacity from rounding
    \item Exploits fragmented slots ILP cannot use
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Why not use smaller $\Delta$?}

\vspace{0.1cm}
\footnotesize
\begin{tabular}{lrr}
\toprule
$\Delta$ & Variables & Solve Time \\
\midrule
1.0s & 5M & 12 min \\
0.5s & 10M & $>$1 hr \\
0.1s & 50M & infeasible \\
\bottomrule
\end{tabular}

\vspace{0.2cm}
\normalsize
\textbf{Conclusion:}\\
ILP is a \textit{pessimistic} oracle for joint routing.\\
CR $<$ 1.0x does not violate theory ---\\
it means the benchmark is loose.\\[6pt]
With $\Delta \to 0$, ILP and PD should converge.
\end{column}
\end{columns}
\end{frame}

\footnotesize
\begin{tabular}{lcc}

\midrule
Histogram & 56\% & 50\% \\

\bottomrule

\vspace{0.15cm}
\normalsize
Histogram is better calibrated, yet EMA routes better. Why?

\vspace{0.2cm}
\textbf{Reason 1: Over-estimation Bias}
\begin{itemize}\setlength{\itemsep}{2pt}
    \item EMA overestimates $\to$ higher perceived value
    \item More willing to use subscription
    \item When quota abundant: beneficial (unused quota = waste)
    \item Automatic aggressive-but-safe policy
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Reason 2: Adaptation Speed}
\begin{itemize}\setlength{\itemsep}{2pt}
    \item EMA adapts in $\sim$\textbf{10 samples}
    \item Histogram needs full window rebuild
    \item Non-stationary workloads $\to$ speed matters more
\end{itemize}

\vspace{0.2cm}
\textbf{2$\times$2 Ablation Results:}

\vspace{0.1cm}
\footnotesize
\begin{tabular}{lcc}
\toprule
 & PD Rule & LA Rule \\
\midrule
EMA & \textbf{CR 1.18} & CR 1.19 \\
Histogram & CR 1.23 & CR 1.22 \\
\bottomrule
\end{tabular}

\vspace{0.15cm}
\normalsize
Switching predictor (row): $\Delta$CR = 0.04\\
Switching rule (column): $\Delta$CR = 0.01\\[6pt]
\alert{Predictor choice $>$ rule complexity}\\[4pt]
\textbf{Takeaway:} Responsiveness trumps calibration.
\end{column}
